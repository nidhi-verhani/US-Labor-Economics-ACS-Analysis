# Load required packages
library(dplyr)
library(rpart)
library(rpart.plot)


# Clean and prepare the data
data <- data %>%
  filter(INCTOT > 0 & INCTOT < 300000) %>%
  mutate(
    IncomeGroup = cut(INCTOT,
                      breaks = c(0, 50000, 100000, 300000),
                      labels = c("Low", "Mid", "High")),
    EDU_SIMPLE = case_when(
      EDUCD %in% c("No schooling completed", "Nursery school, preschool", "Kindergarten") ~ "No Schooling",
      EDUCD %in% c("Grade 1", "Grade 2", "Grade 3", "Grade 4") ~ "Grades 1–4",
      EDUCD %in% c("Grade 5", "Grade 6") ~ "Grades 5–6",
      EDUCD %in% c("Grade 7", "Grade 8") ~ "Grades 7–8",
      EDUCD %in% c("Grade 9", "Grade 10", "Grade 11") ~ "Grades 9–11",
      EDUCD %in% c("12th grade, no diploma", "Regular high school diploma", "GED or alternative credential") ~ "High School",
      grepl("Some college", EDUCD) ~ "Some College",
      grepl("Associate", EDUCD) ~ "Associate Degree",
      grepl("Bachelor", EDUCD) ~ "Bachelor's",
      grepl("Master|Doctoral|Professional", EDUCD) ~ "Grad/Prof Degree",
      TRUE ~ NA_character_
    )
  ) %>%
  filter(!is.na(EDU_SIMPLE), !is.na(SEX), !is.na(UHRSWORK), !is.na(AGE))

# Set factor levels
data$SEX <- as.factor(data$SEX)
data$EDU_SIMPLE <- factor(data$EDU_SIMPLE,
                          levels = c("No Schooling", "Grades 1–4", "Grades 5–6", "Grades 7–8", "Grades 9–11",
                                     "High School", "Some College", "Associate Degree", "Bachelor's", "Grad/Prof Degree"))
data$IncomeGroup <- factor(data$IncomeGroup, levels = c("Low", "Mid", "High"))

# Build the decision tree
tree_model <- rpart(IncomeGroup ~ EDU_SIMPLE + AGE + SEX + UHRSWORK,
                    data = data, method = "class", cp = 0.002, maxdepth = 4)

# Plot with valid color palette for multiclass classification
rpart.plot(tree_model,
           extra = 104,
           box.palette = list("Reds", "Greys", "Greens"),  # Low = red, Mid = grey, High = green
           main = "Income Group Prediction Tree (Education, Age, Gender, Hours)")

#Predicting the Accuracy of the Model

# Split into training and testing sets
set.seed(123)
sample_index <- sample(1:nrow(data), size = 0.7 * nrow(data))
train_data <- data[sample_index, ]
test_data <- data[-sample_index, ]

# Rebuild model on training set
tree_model <- rpart(IncomeGroup ~ EDU_SIMPLE + AGE + SEX + UHRSWORK,
                    data = train_data, method = "class", cp = 0.002, maxdepth = 4)

# Predict on test set
predictions <- predict(tree_model, newdata = test_data, type = "class")

# Create confusion matrix
conf_mat <- table(Predicted = predictions, Actual = test_data$IncomeGroup)

# Print confusion matrix
print(conf_mat)
